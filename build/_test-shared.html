Otopark ücret kademesi

"2 saatten az kalanlar: saati 100 TL. 2–6 saat arası: her saat 40 TL. 6–24 saat arası kalanlar: günlük 250 TL."Hasan Abi, Otopark sahibi (müşteri isteği)

Referans çözüm

public static class Otopark
{
    // dakika: toplam kalış süresi
    public static int UcretHesapla(int dakika)
    {
        if (dakika < 0) throw new ArgumentOutOfRangeException(nameof(dakika));
        if (dakika == 0) return 0;
        int saat = (int)Math.Ceiling(dakika / 60.0);
        if (saat < 2) return saat * 100;
        if (saat <= 6) return saat * 40;
        int gun = (int)Math.Ceiling(dakika / 1440.0);
        return gun * 250; // "6'yı geçen güne girer"
    }
}
